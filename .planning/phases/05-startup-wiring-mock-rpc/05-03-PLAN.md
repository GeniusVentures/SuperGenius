    inline bool UseRealRpcTransport() {
        static bool checked = false;
        static bool use_real = false;
        if (!checked) {
            const char* env = std::getenv("SGNS_E2E_REAL_RPC");
            use_real = (env && std::string(env) == "1");
            checked = true;

    }

    inline TransportFactory GetTransportFactory() {
        if (UseRealRpcTransport()) {
            return [](const std::string& url, std::chrono::seconds timeout) {
                eth::rpc::RpcHttpTransportOptions opts;
                opts.timeout = timeout;
                return std::make_unique<eth::rpc::RpcHttpTransport>(url, opts);
            };
        }
        // Default: mock transport factory (from mock config)
        return CreateMockTransportFactory();
    }
